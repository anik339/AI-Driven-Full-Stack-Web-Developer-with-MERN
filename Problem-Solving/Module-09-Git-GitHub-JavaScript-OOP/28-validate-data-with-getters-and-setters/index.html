Commented JavaScript solution (study after attempting)

Try the problem yourself before opening this reference solution.

function runSolution() {
  class UserProfile {
    #age = 0;

    constructor(name, age) {
      this.name = name;
      // Reuse the setter so constructor input follows the same rules.
      this.age = age;
    }

    get age() {
      return this.#age;
    }

    set age(value) {
      if (!Number.isInteger(value) || value < 0) {
        throw new TypeError("Age must be a non-negative integer.");
      }
      this.#age = value;
    }

    get label() {
      return `${this.name} (${this.#age})`;
    }
  }

  const user = new UserProfile("Anik", 22);
  user.age = 23;
  console.log(user.label);
}
